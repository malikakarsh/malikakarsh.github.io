  function geometry(compact:boolean){
    if(tensor){gl!.deleteBuffer(tensor);buffers.splice(buffers.indexOf(tensor),1);}if(fibers){gl!.deleteBuffer(fibers);buffers.splice(buffers.indexOf(fibers),1);}
    const cells:number[]=[],connections:number[]=[];
    const nx=compact?16:22,ny=compact?7:10,nz=compact?6:8;
    for(let z=0;z<nz;z++)for(let y=0;y<ny;y++)for(let x=0;x<nx;x++){
      const px=(x/(nx-1)-.5)*3.6,py=(y/(ny-1)-.5)*1.65,pz=(z/(nz-1)-.5)*1.44;
      const seed=((x*13+y*7+z*3)%17)/17;
      cells.push(px,py,pz,seed);
      // Fibers expose the depth and continuity of the tensor, without bounding boxes.
      if(x<nx-1 && y%3===0 && z%2===0) connections.push(px,py,pz,seed,px+3.6/(nx-1),py,pz,seed);
      if(z<nz-1 && x%5===0 && y%3===0) connections.push(px,py,pz,seed,px,py,pz+1.44/(nz-1),seed);
    }
    tensor=buffer(cells);fibers=buffer(connections);count=cells.length/4;lineCount=connections.length/4;
  }
